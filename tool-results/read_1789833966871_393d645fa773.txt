    46→function slotGreeting(h: number): { title: string; icon: string; line: string } {
    47→  if (h < 11) return { title: "Good Morning", icon: "🌅", line: "A fresh start — let's make breakfast count." };
    48→  if (h < 16) return { title: "Good Afternoon", icon: "👋", line: "Let's make a healthier you, one meal at a time." };
    49→  if (h < 19) return { title: "Good Evening", icon: "🌤️", line: "Snack o'clock — small bites, chosen well." };
    50→  return { title: "Good Night", icon: "🌙", line: "Dinner time — let's keep the day on track." };
    51→}

    56→  const [now, setNow] = useState<Date | null>(null);

   312→        </div>

   314→          {(["calories", "protein"] as const).map((m) => (

   319→              aria-pressed={metric === m}
   320→              className={cn(
   321→                "px-2.5 py-1 font-medium capitalize transition-colors",
   322→                metric === m ? "bg-primary text-primary-foreground" : "bg-background text-muted-foreground hover:bg-muted",
   323→              )}
   324→            >
   325→              {m}

   327→          ))}
   328→        </div>
   329→      </CardHeader>
   330→      <CardContent>
   331→        <div className="h-44" role="img" aria-label={`Bar chart of ${metric} for the last 7 days with a target line at ${target}`}>
   332→          <TrendBars data={data} target={target} />
   333→        </div>
   334→        <p className="mt-1 text-center text-[11px] text-muted-foreground">

   338→    </Card>

   342→/** Hand-rolled flex bar chart (lighter than a chart lib for this compact card). */
   343→function TrendBars({
   344→  data,
   345→  target,
   346→}: {
   347→  data: { label: string; value: number; onTarget: boolean; isToday: boolean }[];
   348→  target: number;
   349→}) {
   350→  const max = Math.max(target * 1.15, ...data.map((d) => d.value), 1);
   351→  return (
   352→    <div className="relative flex h-full items-end gap-2 pt-3">
   353→      {/* target dashed line */}
   354→      <div
   355→        aria-hidden
   356→        className="absolute inset-x-0 border-t-2 border-dashed border-orange-400/80"
   357→        style={{ bottom: `${(target / max) * 100}%` }}
   358→      >
   359→        <span className="absolute -top-2.5 right-0 rounded bg-orange-400/90 px-1.5 py-0.5 text-[9px] font-semibold text-white">
   360→          Target {target.toLocaleString()}
   361→        </span>
   362→      </div>
   363→      {data.map((d) => {
   364→        const h = Math.max(2, (d.value / max) * 100);
   365→        return (
   366→          <div key={d.label} className="group flex h-full min-w-0 flex-1 flex-col items-center justify-end gap-1">
   367→            <span className="text-[9px] font-medium tabular-nums text-muted-foreground opacity-0 transition-opacity group-hover:opacity-100">
   368→              {d.value.toLocaleString()}
   369→            </span>
   370→            <div
   371→              className={cn(
   372→                "w-full max-w-9 rounded-t-md transition-all duration-500 group-hover:brightness-110",
   373→                d.value === 0
   374→                  ? "bg-muted"
   375→                  : d.isToday
   376→                    ? "bg-gradient-to-t from-emerald-600 to-emerald-400 shadow-md shadow-emerald-600/25"
   377→                    : d.onTarget
   378→                      ? "bg-gradient-to-t from-emerald-500/85 to-emerald-300/85"
   379→                      : "bg-emerald-200/80 dark:bg-emerald-500/35",
   380→              )}
   381→              style={{ height: `${h}%` }}
   382→              title={`${d.label}: ${d.value.toLocaleString()}`}
   383→            />
   384→            <span className={cn("text-[10px]", d.isToday ? "font-bold text-foreground" : "text-muted-foreground")}>{d.label}</span>
   385→          </div>
   386→        );
   387→      })}
   388→    </div>
   389→  );
   390→}

   480→function InsightTipCard({ summary }: { summary: DailySummaryResponse | null }) {
   481→  const violation = summary?.compliance.violations[0];
   482→  const dayIndex = new Date().getDate();

   638→export function HomeView() {
   639→  const dataVersion = useNutriStore((s) => s.dataVersion);
   640→  const [summary, setSummary] = useState<DailySummaryResponse | null>(null);
   641→  const [weekly, setWeekly] = useState<WeeklySummaryResponse | null>(null);
   642→  const [recent, setRecent] = useState<RecentMealsResponse | null>(null);
   643→  const [error, setError] = useState<string | null>(null);
   644→  const [loading, setLoading] = useState(true);
   645→
   646→  const refresh = useCallback(() => {
   647→    let alive = true;
   648→    Promise.all([api.dailySummary(todayKey()), api.weeklySummary(), api.recentMeals()])
   649→      .then(([s, w, r]) => {
   650→        if (!alive) return;
   651→        setSummary(s);
   652→        setWeekly(w);
   653→        setRecent(r);
   654→        setError(null);
   655→        setLoading(false);
   656→      })
   657→      .catch(() => {
   658→        if (!alive) return;
   659→        setError("Could not load your dashboard data.");
   660→        setLoading(false);
   661→      });
   662→    return () => {
   663→      alive = false;
   664→    };
   665→  }, []);
   666→
   667→  useEffect(() => {
   668→    return refresh();
   669→  }, [refresh, dataVersion]);
   670→
   671→  return (
   672→    <div className="space-y-5">
   673→      <div className="grid gap-5 xl:grid-cols-[minmax(0,1fr)_300px]">
   674→        <div className="min-w-0 space-y-5">
   675→          <FadeIn>
   676→            <HeroBanner />
   677→          </FadeIn>
   678→
   679→          {error && (
   680→            <p className="rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-sm text-destructive" role="alert">
   681→              {error}
   682→            </p>
   683→          )}
   684→
   685→          {loading && !summary && (
   686→            <div className="space-y-5" aria-hidden>
   687→              <Skeleton className="h-44 w-full rounded-3xl" />
   688→              <Skeleton className="h-40 w-full rounded-3xl" />
   689→            </div>
   690→          )}
   691→
   692→          {summary && (
   693→            <FadeIn delay={0.05}>
   694→              <TodaysNutrition summary={summary} />
   695→            </FadeIn>
   696→          )}
   697→
   698→          <div className="grid gap-5 lg:grid-cols-2">
   699→            {weekly ? (
   700→              <FadeIn delay={0.1}>
   701→                <TrendChart weekly={weekly} />
   702→              </FadeIn>
   703→            ) : (
   704→              <Skeleton className="h-64 rounded-2xl" aria-hidden />
   705→            )}
   706→            <FadeIn delay={0.12}>
   707→              <RecommendationCard />
   708→            </FadeIn>
   709→          </div>
   710→
   711→          <div className="grid gap-5 lg:grid-cols-2">

   715→              </FadeIn>

   717→              <Skeleton className="h-56 rounded-2xl" aria-hidden />
