    99→    99→function draftFromServer(data: ProfileResponse): Draft {
   100→   100→  const p = data.profile;
   101→   101→  return {
   102→   102→    age: p.age?.toString() ?? "",
   103→   103→    sex: p.sex ? p.sex.charAt(0).toUpperCase() + p.sex.slice(1).toLowerCase() : "Male",
   104→   104→    heightCm: p.heightCm?.toString() ?? "",
   105→   105→    weightKg: p.weightKg?.toString() ?? "",
   106→   106→    activityLevel: p.activityLevel ?? "moderate",
   107→   107→    goal: p.goal ?? "maintain",
   108→   108→    dietaryPreference: p.dietaryPreference ?? "vegetarian",
   109→   109→    allergies: p.allergies ?? [],
   110→   110→    healthConditions: p.healthConditions ?? [],
   111→   111→    language: p.language ?? "en",
   112→   112→    regional: (p.language ?? "en") !== "en",
   113→   113→  };
   114→   114→}
   115→   115→
   116→   116→/** Server stores sex lowercase — the UI shows display case. */
   117→   117→function sexForServer(display: string): string {
   118→   118→  return display.toLowerCase();
   119→   119→}
   120→   120→
   121→   121→function initialsOf(name: string): string {
   122→   122→  return name
   123→   123→    .split(/\s+/)
   124→   124→    .filter(Boolean)
   125→   125→    .map((w) => w[0])
   126→   126→    .slice(0, 2)
   127→   127→    .join("")
   128→   128→    .toUpperCase();
   129→   129→}
