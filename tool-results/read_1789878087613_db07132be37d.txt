   159→   159→/** Human relative time for recent-input chips ("2 minutes ago", "Yesterday"). */
   160→   160→function relTime(iso: string): string {
   161→   161→  const then = new Date(iso).getTime();
   162→   162→  if (!Number.isFinite(then)) return "";
   163→   163→  const mins = Math.max(0, Math.round((Date.now() - then) / 60000));
   164→   164→  if (mins < 1) return "just now";
   165→   165→  if (mins < 60) return `${mins} minute${mins === 1 ? "" : "s"} ago`;
   166→   166→  const thenDay = new Date(iso);
   167→   167→  const now = new Date();
   168→   168→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   169→   169→  const startOfThat = new Date(thenDay.getFullYear(), thenDay.getMonth(), thenDay.getDate()).getTime();
   170→   170→  const dayDiff = Math.round((startOfToday - startOfThat) / 86400000);
   171→   171→  if (dayDiff === 0) {
   172→   172→    const hours = Math.round(mins / 60);
   173→   173→    return `${hours} hour${hours === 1 ? "" : "s"} ago`;
   174→   174→  }
   175→   175→  if (dayDiff === 1) return "Yesterday";
   176→   176→  if (dayDiff < 7) return `${dayDiff} days ago`;
   177→   177→  return thenDay.toLocaleDateString("en-GB", { day: "numeric", month: "short" });
   178→   178→}

   396→   396→          lineId: f.lineId,
   397→   397→          originalName: f.originalName,
   398→   398→          displayName: f.displayName,
   399→   399→          foodId: f.foodId,
   400→   400→          matchStatus: f.matchStatus,
   401→   401→          candidates: f.candidates,
   402→   402→          quantity: f.quantity,
   403→   403→          unit: f.unit,
   404→   404→          preparation: f.preparation ?? null,
   405→   405→          confidence: f.confidence,
   406→   406→          quantitySource: f.quantitySource,

   410→   410→      setConfirmResult(null);
   411→   411→      setStep("review");

   421→   421→    } finally {
   422→   422→      setAnalyzing(false);
   423→   423→    }
