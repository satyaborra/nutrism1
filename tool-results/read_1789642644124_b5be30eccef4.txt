    91→    91→function EditMealDialog({
    92→    92→  meal,
    93→    93→  open,
    94→    94→  onOpenChange,
    95→    95→  onSaved,
    96→    96→}: {
    97→    97→  meal: MealDetail | null;
    98→    98→  open: boolean;
    99→    99→  onOpenChange: (open: boolean) => void;
   100→   100→  onSaved: () => void;
   101→   101→}) {
   102→   102→  const { toast } = useToast();
   103→   103→  const [lines, setLines] = useState<Record<string, EditLineState>>({});
   104→   104→  const [saving, setSaving] = useState(false);
   105→   105→
   106→   106→  useEffect(() => {
   107→   107→    if (meal && open) {
   108→   108→      const init: Record<string, EditLineState> = {};
   109→   109→      for (const f of meal.foods) init[f.id] = { qty: String(f.quantity), unit: f.unit, removed: false };
   110→   110→      setLines(init);
   111→   111→    }
   112→   112→  }, [meal, open]);
   113→   113→
   114→   114→  const changePreview = useMemo(() => {
   115→   115→    if (!meal) return { any: false, approxKcal: 0 };
   116→   116→    let approx = 0;
   117→   117→    let any = false;
   118→   118→    for (const f of meal.foods) {
   119→   119→      const st = lines[f.id];
   120→   120→      if (!st) continue;
   121→   121→      if (st.removed) {
   122→   122→        any = true;
   123→   123→        continue;
   124→   124→      }
   125→   125→      const q = Number(st.qty);
   126→   126→      if (!Number.isFinite(q) || q <= 0) continue;
   127→   127→      if (q !== f.quantity || st.unit !== f.unit) any = true;
   128→   128→      // Rough client preview only — the server recalculates exact values from the Food table.
   129→   129→      approx += (f.nutrition.calories ?? 0) * (q / f.quantity);
   130→   130→    }
   131→   131→    return { any, approxKcal: approx };
   132→   132→  }, [lines, meal]);
   133→   133→
   134→   134→  if (!meal) return null;
   135→   135→
   136→   136→  function setLine(id: string, patch: Partial<EditLineState>) {
   137→   137→    setLines((prev) => ({ ...prev, [id]: { ...prev[id], ...patch } }));
   138→   138→  }
   139→   139→
   140→   140→  async function handleSave() {
   141→   141→    if (!meal) return;
   142→   142→    const built: { lineId: string; remove?: true; quantity?: number; unit?: string }[] = [];
   143→   143→    let invalid = false;
   144→   144→    for (const f of meal.foods) {
   145→   145→      const st = lines[f.id];
   146→   146→      if (!st) continue;
   147→   147→      if (st.removed) {
   148→   148→        built.push({ lineId: f.id, remove: true });
   149→   149→        continue;
   150→   150→      }
   151→   151→      const q = Number(st.qty);
   152→   152→      if (!Number.isFinite(q) || q <= 0) {
   153→   153→        invalid = true;
   154→   154→        break;
   155→   155→      }
   156→   156→      if (q !== f.quantity || st.unit !== f.unit) built.push({ lineId: f.id, quantity: q, unit: st.unit });
   157→   157→    }
   158→   158→
   159→   159→    if (invalid) {
   160→   160→      toast({ title: "Check quantities", description: "Every kept line needs a quantity above 0.", variant: "destructive" });
   161→   161→      return;
   162→   162→    }
   163→   163→    if (built.length === 0) {
   164→   164→      toast({ title: "Nothing changed", description: "Adjust a quantity, switch a unit, or remove a line." });
   165→   165→      return;
   166→   166→    }
   167→   167→    const removedCount = built.filter((f) => f.remove).length;
   168→   168→    if (meal.foods.length - removedCount === 0) {
   169→   169→      toast({ title: "Cannot remove everything", description: "A meal needs at least one item — delete the meal instead.", variant: "destructive" });
   170→   170→      return;
   171→   171→    }
   172→   172→
   173→   173→    setSaving(true);
   174→   174→    try {
   175→   175→      const res = await api.editMeal(meal.id, { foods: built });
   176→   176→      const convNote = res.conversionNotes?.[0]?.note;
   177→   177→      toast({
   178→   178→        title: "Meal updated",
   179→   179→        description: convNote
   180→   180→          ? `${formatKcal(res.previousCalories)} → ${formatKcal(res.totals.calories)} · ${convNote}`
   181→   181→          : `${formatKcal(res.previousCalories)} → ${formatKcal(res.totals.calories)} · recalculated from the food database.`,
   182→   182→      });
   183→   183→      onOpenChange(false);
   184→   184→      onSaved();
   185→   185→    } catch (e) {
   186→   186→      toast({
   187→   187→        title: "Could not update meal",
   188→   188→        description: e instanceof Error ? e.message : "Please try again.",
   189→   189→        variant: "destructive",
   190→   190→      });
   191→   191→    } finally {
   192→   192→      setSaving(false);
   193→   193→    }
   194→   194→  }
   195→   195→
   196→   196→  const unitsFor = (current: string) => Array.from(new Set([current, ...EDIT_UNITS]));
   197→   197→
   198→   198→  return (
   199→   199→    <Dialog open={open} onOpenChange={(o) => !saving && onOpenChange(o)}>
   200→   200→      <DialogContent className="sm:max-w-md">
   201→   201→        <DialogHeader>
   202→   202→          <DialogTitle className="flex items-center gap-2">
   203→   203→            <Pencil className="h-4 w-4 text-primary" aria-hidden />
   204→   204→            Edit {mealLabel(meal.mealType).toLowerCase()}
   205→   205→          </DialogTitle>
   206→   206→          <DialogDescription>
   207→   207→            Adjust quantities or units — totals are recalculated server-side from the food database, never from this form.
   208→   208→          </DialogDescription>
   209→   209→        </DialogHeader>
   210→   210→
   211→   211→        <div className="max-h-[55vh] space-y-2 overflow-y-auto pr-1 [scrollbar-width:thin]">
   212→   212→          {meal.foods.map((f) => {
   213→   213→            const st = lines[f.id];
   214→   214→            if (!st) return null;
   215→   215→            const qNum = Number(st.qty);
   216→   216→            const approx = Number.isFinite(qNum) && qNum > 0 ? (f.nutrition.calories ?? 0) * (qNum / f.quantity) : 0;
   217→   217→            const changed = qNum !== f.quantity || st.unit !== f.unit;
   218→   218→            return (
   219→   219→              <div
   220→   220→                key={f.id}
   221→   221→                className={cn(
   222→   222→                  "rounded-lg border p-2.5 transition-opacity",
   223→   223→                  st.removed ? "border-destructive/30 bg-destructive/5 opacity-60" : "bg-muted/30",
   224→   224→                )}
   225→   225→              >
   226→   226→                <div className="flex items-center justify-between gap-2">
   227→   227→                  <p className="min-w-0 truncate text-sm font-medium">{f.name}</p>
   228→   228→                  <Button
   229→   229→                    variant="ghost"
   230→   230→                    size="sm"
   231→   231→                    className={cn("h-7 px-2 text-xs", st.removed ? "text-foreground" : "text-muted-foreground hover:text-destructive")}
   232→   232→                    onClick={() => setLine(f.id, { removed: !st.removed })}
   233→   233→                  >
   234→   234→                    {st.removed ? "Keep" : "Remove"}
   235→   235→                  </Button>
   236→   236→                </div>
   237→   237→                <div className="mt-2 flex items-end gap-2">
   238→   238→                  <div className="w-24 space-y-1">
   239→   239→                    <Label htmlFor={`qty-${f.id}`} className="text-[10px] uppercase tracking-wide text-muted-foreground">Quantity</Label>
   240→   240→                    <Input
   241→   241→                      id={`qty-${f.id}`}
   242→   242→                      type="number"
   243→   243→                      inputMode="decimal"
   244→   244→                      min={0.1}
   245→   245→                      max={100}
   246→   246→                      step={0.5}
   247→   247→                      value={st.qty}
   248→   248→                      disabled={st.removed}
   249→   249→                      onChange={(e) => setLine(f.id, { qty: e.target.value })}
   250→   250→                      className="h-8 tabular-nums"
   251→   251→                    />
   252→   252→                  </div>
   253→   253→                  <div className="flex-1 space-y-1">
   254→   254→                    <Label htmlFor={`unit-${f.id}`} className="text-[10px] uppercase tracking-wide text-muted-foreground">Unit</Label>
   255→   255→                    <Select value={st.unit} onValueChange={(v) => setLine(f.id, { unit: v })} disabled={st.removed}>
   256→   256→                      <SelectTrigger id={`unit-${f.id}`} className="h-8">
   257→   257→                        <SelectValue />
   258→   258→                      </SelectTrigger>
   259→   259→                      <SelectContent>
   260→   260→                        {unitsFor(f.unit).map((u) => (
   261→   261→                          <SelectItem key={u} value={u}>{u}</SelectItem>
   262→   262→                        ))}
   263→   263→                      </SelectContent>
   264→   264→                    </Select>
   265→   265→                  </div>
   266→   266→                  <p className={cn("pb-1.5 text-right text-xs tabular-nums", changed ? "font-semibold text-primary" : "text-muted-foreground")}>
   267→   267→                    ≈{Math.round(approx)} kcal
   268→   268→                  </p>
   269→   269→                </div>
   270→   270→                {changed && !st.removed && (
   271→   271→                  <p className="mt-1.5 text-[10px] text-muted-foreground">
   272→   272→                    Was {f.quantity} {f.unit} · {formatKcal(f.nutrition.calories ?? 0)} — preview is approximate, server recalculates.
   273→   273→                  </p>
   274→   274→                )}
   275→   275→              </div>
   276→   276→            );
   277→   277→          })}
   278→   278→        </div>
   279→   279→
   280→   280→        <div className="rounded-lg border border-primary/25 bg-primary/5 px-3 py-2 text-xs">
   281→   281→          <span className="text-muted-foreground">New meal total (preview): </span>
   282→   282→          <span className="font-semibold tabular-nums">≈{Math.round(changePreview.approxKcal)} kcal</span>
   283→   283→          <span className="text-muted-foreground"> · exact values computed on save</span>
   284→   284→        </div>
   285→   285→
   286→   286→        <DialogFooter>
   287→   287→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>Cancel</Button>
   288→   288→          <Button onClick={() => void handleSave()} disabled={saving || !changePreview.any}>
   289→   289→            {saving ? (
   290→   290→              <>
   291→   291→                <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden /> Recalculating…
   292→   292→              </>
   293→   293→            ) : (
   294→   294→              "Save changes"
   295→   295→            )}
   296→   296→          </Button>
   297→   297→        </DialogFooter>
   298→   298→      </DialogContent>
   299→   299→    </Dialog>
   300→   300→  );
   301→   301→}

   303→   303→export function RecentMeals() {

   320→   320→      .then((res) => alive && (setData(res), setError(null)))
   321→   321→      .catch(() => alive && setError("Could not load your meals."));
   322→   322→    return () => {
   323→   323→      alive = false;
   324→   324→    };
   325→   325→  }, [dataVersion]);
   326→   326→
   327→   327→  async function handleDelete() {

   335→   335→      });
   336→   336→      setPendingDelete(null);
   337→   337→      bumpData();
   338→   338→    } catch (e) {
   339→   339→      toast({
   340→   340→        title: "Could not delete meal",
   341→   341→        description: e instanceof Error ? e.message : "Please try again.",
   342→   342→        variant: "destructive",
   343→   343→      });
   344→   344→    } finally {
   345→   345→      setDeleting(false);
   346→   346→    }

   355→   355→        title: "Logged again",

   357→   357→      });
   358→   358→      setPendingRelog(null);
   359→   359→      bumpData();
   360→   360→    } catch (e) {
   361→   361→      toast({
   362→   362→        title: "Could not log this meal again",
   363→   363→        description: e instanceof Error ? e.message : "Please try again.",
   364→   364→        variant: "destructive",
   365→   365→      });
   366→   366→    } finally {
   367→   367→      setRelogging(false);
   368→   368→    }
   369→   369→  }

   371→   371→  return (
   372→   372→    <Card className="transition-shadow duration-300 hover:shadow-md hover:shadow-primary/5">
   373→   373→      <CardHeader>
   374→   374→        <CardTitle className="flex items-center gap-2">

   376→   376→          Recent meals
   377→   377→        </CardTitle>
   378→   378→        <CardDescription>
   379→   379→          Your last logged meals with server-verified totals. Expand for the full nutrient panel, or edit and delete a mistaken entry.
   380→   380→        </CardDescription>
   381→   381→      </CardHeader>
   382→   382→      <CardContent>
   383→   383→        {error && <p className="rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-sm text-destructive">{error}</p>}
   384→   384→
   385→   385→        {!data && !error && (
   386→   386→          <div className="space-y-2">
   387→   387→            <Skeleton className="h-16 w-full" />
   388→   388→            <Skeleton className="h-16 w-full" />
   389→   389→            <Skeleton className="h-16 w-full" />
   390→   390→          </div>
   391→   391→        )}
   392→   392→
   393→   393→        {data && data.count === 0 && (
   394→   394→          <p className="rounded-xl border border-dashed p-6 text-center text-sm text-muted-foreground">
   395→   395→            Nothing logged yet — log your first meal above and it will appear here.
   396→   396→          </p>
   397→   397→        )}
   398→   398→
   399→   399→        {data && data.count > 0 && (
   400→   400→          <ul className="max-h-96 space-y-2 overflow-y-auto pr-1 [scrollbar-width:thin]">
   401→   401→            {data.meals.map((m) => (
   402→   402→              <li key={m.id}>
   403→   403→                <Collapsible>
   404→   404→                  <div className="group/meal relative rounded-xl border transition-colors hover:border-primary/40 hover:bg-muted/30 data-[state=open]:border-primary/40">
   405→   405→                    <CollapsibleTrigger className="flex w-full items-center gap-3 rounded-xl p-3 pr-20 text-left transition-colors hover:bg-muted/50">
   406→   406→                      <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-muted text-base transition-transform group-hover/meal:scale-105" aria-hidden>
   407→   407→                        {MEAL_TYPE_ICON[m.mealType] ?? "🍽️"}
   408→   408→                      </span>
   409→   409→                      <span className="min-w-0 flex-1">
