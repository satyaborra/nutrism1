   121→function dayKeyOf(d: Date): string {
   122→  const y = d.getFullYear();
   123→  const m = String(d.getMonth() + 1).padStart(2, "0");
   124→  const day = String(d.getDate()).padStart(2, "0");
   125→  return `${y}-${m}-${day}`;
   126→}

   159→/** Human relative time for recent-input chips ("2 minutes ago", "Yesterday"). */
   160→function relTime(iso: string): string {
   161→  const then = new Date(iso).getTime();
   162→  if (!Number.isFinite(then)) return "";
   163→  const mins = Math.max(0, Math.round((Date.now() - then) / 60000));
   164→  if (mins < 1) return "just now";
   165→  if (mins < 60) return `${mins} minute${mins === 1 ? "" : "s"} ago`;
   166→  const thenDay = new Date(iso);
   167→  const now = new Date();
   168→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   169→  const startOfThat = new Date(thenDay.getFullYear(), thenDay.getMonth(), thenDay.getDate()).getTime();
   170→  const dayDiff = Math.round((startOfToday - startOfThat) / 86400000);
   171→  if (dayDiff === 0) {
   172→    const hours = Math.round(mins / 60);
   173→    return `${hours} hour${hours === 1 ? "" : "s"} ago`;
   174→  }
   175→  if (dayDiff === 1) return "Yesterday";
   176→  if (dayDiff < 7) return `${dayDiff} days ago`;
   177→  return thenDay.toLocaleDateString("en-GB", { day: "numeric", month: "short" });
   178→}

   322→  /** Voice input via the browser Web Speech API — transcript lands in the

   324→  function startVoiceInput() {
   325→    const Ctor = getSpeechRecognition();
   326→    if (!Ctor) {
   327→      toast({
   328→        title: "Voice input not available",
   329→        description: "Your browser doesn't support speech recognition — try Chrome, or type the meal instead.",
   330→        variant: "destructive",
   331→      });
   332→      return;
   333→    }
   334→    if (voiceRef.current) {
   335→      voiceRef.current.stop();
   336→      voiceRef.current = null;
   337→      setVoiceListening(false);
   338→      return;
   339→    }
   340→    try {
   341→      const rec = new Ctor();
   342→      rec.lang = VOICE_LOCALES[voiceLang ?? "en"] ?? "en-IN";
   343→      rec.interimResults = false;
   344→      rec.continuous = false;
   345→      rec.onresult = (e) => {
   346→        const transcript = Array.from({ length: e.results.length }, (_, i) => e.results[i][0]?.transcript ?? "")
   347→          .join(" ")
   348→          .trim();

   351→          setMode("text");

   358→        voiceRef.current = null;
   359→      };

   361→        setVoiceListening(false);

   375→  async function handleAnalyze() {
   376→    setAnalyzing(true);
   377→    setAnalyzeError(null);

   379→      const payload: { text?: string; imageDataUrl?: string; hint?: string } = {};

   387→        draftId: res.draftId,
   388→        inputType: mode === "text" ? "text" : "image",
   389→        detectedLanguage: res.detectedLanguage,
   390→        aiOk: res.aiOk,
   391→        aiNote: res.aiNote,
   392→        aiLatencyMs: res.aiLatencyMs,

   395→        res.foods.map((f) => ({
   396→          lineId: f.lineId,
   397→          originalName: f.originalName,
   398→          displayName: f.displayName,
   399→          foodId: f.foodId,
   400→          matchStatus: f.matchStatus,
   401→          candidates: f.candidates,

   403→          unit: f.unit,

   405→          confidence: f.confidence,
   406→          quantitySource: f.quantitySource,
   407→        }))
   408→      );
   409→      setMealType(res.mealTypeGuess || "snack");
   410→      setConfirmResult(null);
   411→      setStep("review");
   412→      if (!res.aiOk) {
   413→        toast({
   414→          title: "Parsed offline",

   420→      setAnalyzeError(e instanceof ApiError ? e.message : "Analysis failed. Please try again.");

   504→        ...(isBackfill ? { eatenAt: `${logDate}T${SLOT_DEFAULT_TIME[mealType] ?? "12:30"}:00` } : {}),

   510→      onLogged();

   512→        mealType,
   513→        duplicate: res.duplicate,
   514→        totals: res.totals,

   525→    } finally {
   526→      setLogging(false);
   527→    }
