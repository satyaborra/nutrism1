    80→    80→type FieldKey = "age" | "heightCm" | "weightKg";
    81→    81→
    82→    82→const FIELD_LIMITS: Record<FieldKey, { min: number; max: number; message: string }> = {
    83→    83→  age: { min: 5, max: 120, message: "Age must be between 5 and 120." },
    84→    84→  heightCm: { min: 80, max: 250, message: "Height must be between 80 and 250 cm." },
    85→    85→  weightKg: { min: 15, max: 400, message: "Weight must be between 15 and 400 kg." },

    88→    88→const FIELD_ORDER: FieldKey[] = ["age", "heightCm", "weightKg"];
    89→    89→
    90→    90→/** Keep digits and a single decimal dot; drop everything else. */
    91→    91→function sanitizeNumeric(v: string): string {
    92→    92→  let out = "";
    93→    93→  let dot = false;
    94→    94→  for (const ch of v) {
    95→    95→    if (ch >= "0" && ch <= "9") out += ch;
    96→    96→    else if (ch === "." && !dot) {
    97→    97→      out += ".";
    98→    98→      dot = true;
    99→    99→    }
   100→   100→  }
   101→   101→  return out;
   102→   102→}
   103→   103→
   104→   104→function validateDraft(draft: Record<FieldKey, string>): Partial<Record<FieldKey, string>> {
   105→   105→  const errors: Partial<Record<FieldKey, string>> = {};
   106→   106→  for (const key of FIELD_ORDER) {
   107→   107→    const raw = draft[key].trim();
   108→   108→    if (raw === "") continue; // empty = clear the value (null is allowed server-side)
   109→   109→    const n = Number(raw);
   110→   110→    if (!Number.isFinite(n) || n < FIELD_LIMITS[key].min || n > FIELD_LIMITS[key].max) {
   111→   111→      errors[key] = FIELD_LIMITS[key].message;
   112→   112→    }
   113→   113→  }
   114→   114→  return errors;
   115→   115→}
