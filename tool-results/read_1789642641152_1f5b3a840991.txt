    51→function formatValue(n: number, decimals: number): string {
    52→  if (decimals === 0) return String(Math.round(n));
    53→  return (Math.round((n + Number.EPSILON) * 10) / 10).toFixed(1);
    54→}

    56→/** Compact per-meal compliance badge — states evaluated server-side from DB constraints. */
    57→function ComplianceBadge({ compliance, conditions }: { compliance?: Compliance; conditions: string[] }) {
    58→  if (!compliance || conditions.length === 0) return null;
    59→  if (compliance.state === "COMPLIANT") {
    60→    return (
    61→      <Badge variant="outline" className="gap-1 border-emerald-500/40 bg-emerald-500/10 px-1.5 py-0 text-[9px] uppercase tracking-wide text-emerald-700 dark:text-emerald-400">
    62→        <CheckCircle2 className="h-2.5 w-2.5" aria-hidden /> ok
    63→      </Badge>
    64→    );
    65→  }
    66→  if (compliance.state === "POTENTIALLY_NON_COMPLIANT") {
    67→    const tip = compliance.violations.map((v) => `${v.condition}: ${v.message}`).join("\n");
    68→    return (
    69→      <Badge
    70→        variant="outline"
    71→        title={tip}
    72→        className="gap-1 border-amber-500/40 bg-amber-500/10 px-1.5 py-0 text-[9px] uppercase tracking-wide text-amber-700 dark:text-amber-400"
    73→      >
    74→        <TriangleAlert className="h-2.5 w-2.5" aria-hidden /> watch
    75→      </Badge>
    76→    );
    77→  }
    78→  return (
    79→    <Badge variant="outline" title="Not all values were available to verify your conditions." className="gap-1 px-1.5 py-0 text-[9px] uppercase tracking-wide text-muted-foreground">
    80→      <HelpCircle className="h-2.5 w-2.5" aria-hidden /> unclear
    81→    </Badge>
    82→  );
    83→}

    91→function EditMealDialog({
    92→  meal,
    93→  open,
    94→  onOpenChange,
    95→  onSaved,
    96→}: {
    97→  meal: MealDetail | null;
    98→  open: boolean;
    99→  onOpenChange: (open: boolean) => void;
   100→  onSaved: () => void;
   101→}) {
   102→  const { toast } = useToast();
   103→  const [lines, setLines] = useState<Record<string, EditLineState>>({});
   104→  const [saving, setSaving] = useState(false);
   105→
   106→  useEffect(() => {
   107→    if (meal && open) {
   108→      const init: Record<string, EditLineState> = {};
   109→      for (const f of meal.foods) init[f.id] = { qty: String(f.quantity), unit: f.unit, removed: false };
   110→      setLines(init);
   111→    }
   112→  }, [meal, open]);
   113→
   114→  const changePreview = useMemo(() => {
   115→    if (!meal) return { any: false, approxKcal: 0 };
   116→    let approx = 0;
   117→    let any = false;
   118→    for (const f of meal.foods) {
   119→      const st = lines[f.id];
   120→      if (!st) continue;
   121→      if (st.removed) {
   122→        any = true;
   123→        continue;
   124→      }
   125→      const q = Number(st.qty);
   126→      if (!Number.isFinite(q) || q <= 0) continue;
   127→      if (q !== f.quantity || st.unit !== f.unit) any = true;
   128→      // Rough client preview only — the server recalculates exact values from the Food table.
   129→      approx += (f.nutrition.calories ?? 0) * (q / f.quantity);
   130→    }
   131→    return { any, approxKcal: approx };
   132→  }, [lines, meal]);
   133→
   134→  if (!meal) return null;
   135→
   136→  function setLine(id: string, patch: Partial<EditLineState>) {
   137→    setLines((prev) => ({ ...prev, [id]: { ...prev[id], ...patch } }));
   138→  }
   139→
   140→  async function handleSave() {
   141→    if (!meal) return;
   142→    const built: { lineId: string; remove?: true; quantity?: number; unit?: string }[] = [];
   143→    let invalid = false;
   144→    for (const f of meal.foods) {
   145→      const st = lines[f.id];
   146→      if (!st) continue;
   147→      if (st.removed) {
   148→        built.push({ lineId: f.id, remove: true });
   149→        continue;
   150→      }
   151→      const q = Number(st.qty);
   152→      if (!Number.isFinite(q) || q <= 0) {
   153→        invalid = true;
   154→        break;
   155→      }
   156→      if (q !== f.quantity || st.unit !== f.unit) built.push({ lineId: f.id, quantity: q, unit: st.unit });
   157→    }
   158→
   159→    if (invalid) {
   160→      toast({ title: "Check quantities", description: "Every kept line needs a quantity above 0.", variant: "destructive" });
   161→      return;
   162→    }
   163→    if (built.length === 0) {
   164→      toast({ title: "Nothing changed", description: "Adjust a quantity, switch a unit, or remove a line." });
   165→      return;
   166→    }
   167→    const removedCount = built.filter((f) => f.remove).length;
   168→    if (meal.foods.length - removedCount === 0) {
   169→      toast({ title: "Cannot remove everything", description: "A meal needs at least one item — delete the meal instead.", variant: "destructive" });
   170→      return;
   171→    }
   172→
   173→    setSaving(true);
   174→    try {
   175→      const res = await api.editMeal(meal.id, { foods: built });
   176→      const convNote = res.conversionNotes?.[0]?.note;
   177→      toast({
   178→        title: "Meal updated",
   179→        description: convNote
   180→          ? `${formatKcal(res.previousCalories)} → ${formatKcal(res.totals.calories)} · ${convNote}`
   181→          : `${formatKcal(res.previousCalories)} → ${formatKcal(res.totals.calories)} · recalculated from the food database.`,
   182→      });
   183→      onOpenChange(false);
   184→      onSaved();
   185→    } catch (e) {
   186→      toast({
   187→        title: "Could not update meal",
   188→        description: e instanceof Error ? e.message : "Please try again.",
   189→        variant: "destructive",
   190→      });
   191→    } finally {
   192→      setSaving(false);
   193→    }
   194→  }
   195→
   196→  const unitsFor = (current: string) => Array.from(new Set([current, ...EDIT_UNITS]));
   197→
   198→  return (
   199→    <Dialog open={open} onOpenChange={(o) => !saving && onOpenChange(o)}>
   200→      <DialogContent className="sm:max-w-md">
   201→        <DialogHeader>
   202→          <DialogTitle className="flex items-center gap-2">
   203→            <Pencil className="h-4 w-4 text-primary" aria-hidden />
   204→            Edit {mealLabel(meal.mealType).toLowerCase()}
   205→          </DialogTitle>
   206→          <DialogDescription>
   207→            Adjust quantities or units — totals are recalculated server-side from the food database, never from this form.
   208→          </DialogDescription>
   209→        </DialogHeader>
   210→
   211→        <div className="max-h-[55vh] space-y-2 overflow-y-auto pr-1 [scrollbar-width:thin]">
   212→          {meal.foods.map((f) => {
   213→            const st = lines[f.id];
   214→            if (!st) return null;
   215→            const qNum = Number(st.qty);
   216→            const approx = Number.isFinite(qNum) && qNum > 0 ? (f.nutrition.calories ?? 0) * (qNum / f.quantity) : 0;
   217→            const changed = qNum !== f.quantity || st.unit !== f.unit;
   218→            return (
   219→              <div
   220→                key={f.id}
   221→                className={cn(
   222→                  "rounded-lg border p-2.5 transition-opacity",
   223→                  st.removed ? "border-destructive/30 bg-destructive/5 opacity-60" : "bg-muted/30",
   224→                )}
   225→              >
   226→                <div className="flex items-center justify-between gap-2">
   227→                  <p className="min-w-0 truncate text-sm font-medium">{f.name}</p>
   228→                  <Button
   229→                    variant="ghost"
   230→                    size="sm"
   231→                    className={cn("h-7 px-2 text-xs", st.removed ? "text-foreground" : "text-muted-foreground hover:text-destructive")}
   232→                    onClick={() => setLine(f.id, { removed: !st.removed })}
   233→                  >
   234→                    {st.removed ? "Keep" : "Remove"}
   235→                  </Button>
   236→                </div>
   237→                <div className="mt-2 flex items-end gap-2">
   238→                  <div className="w-24 space-y-1">
   239→                    <Label htmlFor={`qty-${f.id}`} className="text-[10px] uppercase tracking-wide text-muted-foreground">Quantity</Label>
   240→                    <Input
   241→                      id={`qty-${f.id}`}
   242→                      type="number"
   243→                      inputMode="decimal"
   244→                      min={0.1}
   245→                      max={100}
   246→                      step={0.5}
   247→                      value={st.qty}
   248→                      disabled={st.removed}
   249→                      onChange={(e) => setLine(f.id, { qty: e.target.value })}
   250→                      className="h-8 tabular-nums"
   251→                    />
   252→                  </div>
   253→                  <div className="flex-1 space-y-1">
   254→                    <Label htmlFor={`unit-${f.id}`} className="text-[10px] uppercase tracking-wide text-muted-foreground">Unit</Label>
   255→                    <Select value={st.unit} onValueChange={(v) => setLine(f.id, { unit: v })} disabled={st.removed}>
   256→                      <SelectTrigger id={`unit-${f.id}`} className="h-8">
   257→                        <SelectValue />
   258→                      </SelectTrigger>
   259→                      <SelectContent>
   260→                        {unitsFor(f.unit).map((u) => (
   261→                          <SelectItem key={u} value={u}>{u}</SelectItem>
   262→                        ))}
   263→                      </SelectContent>
   264→                    </Select>
   265→                  </div>
   266→                  <p className={cn("pb-1.5 text-right text-xs tabular-nums", changed ? "font-semibold text-primary" : "text-muted-foreground")}>
   267→                    ≈{Math.round(approx)} kcal
   268→                  </p>
   269→                </div>
   270→                {changed && !st.removed && (
   271→                  <p className="mt-1.5 text-[10px] text-muted-foreground">
   272→                    Was {f.quantity} {f.unit} · {formatKcal(f.nutrition.calories ?? 0)} — preview is approximate, server recalculates.
   273→                  </p>
   274→                )}
   275→              </div>
   276→            );
   277→          })}
   278→        </div>
   279→
   280→        <div className="rounded-lg border border-primary/25 bg-primary/5 px-3 py-2 text-xs">
   281→          <span className="text-muted-foreground">New meal total (preview): </span>
   282→          <span className="font-semibold tabular-nums">≈{Math.round(changePreview.approxKcal)} kcal</span>
   283→          <span className="text-muted-foreground"> · exact values computed on save</span>
   284→        </div>
   285→
   286→        <DialogFooter>
   287→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>Cancel</Button>
   288→          <Button onClick={() => void handleSave()} disabled={saving || !changePreview.any}>
   289→            {saving ? (
   290→              <>
   291→                <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden /> Recalculating…
   292→              </>
   293→            ) : (
   294→              "Save changes"
   295→            )}
   296→          </Button>
   297→        </DialogFooter>
   298→      </DialogContent>
   299→    </Dialog>
   300→  );
   301→}

   303→export function RecentMeals() {
   304→  const dataVersion = useNutriStore((s) => s.dataVersion);
   305→  const bumpData = useNutriStore((s) => s.bumpData);
   306→  const healthConditions = useNutriStore((s) => s.profileBrief?.healthConditions ?? []);
   307→  const { toast } = useToast();
   308→  const [data, setData] = useState<RecentMealsResponse | null>(null);
   309→  const [error, setError] = useState<string | null>(null);
   310→  const [pendingDelete, setPendingDelete] = useState<MealDetail | null>(null);
   311→  const [deleting, setDeleting] = useState(false);
   312→  const [editing, setEditing] = useState<MealDetail | null>(null);
   313→  const [pendingRelog, setPendingRelog] = useState<MealDetail | null>(null);
   314→  const [relogging, setRelogging] = useState(false);
   315→
   316→  useEffect(() => {
   317→    let alive = true;
   318→    api
   319→      .recentMeals()
   320→      .then((res) => alive && (setData(res), setError(null)))
   321→      .catch(() => alive && setError("Could not load your meals."));
   322→    return () => {
   323→      alive = false;
   324→    };
   325→  }, [dataVersion]);
   326→
   327→  async function handleDelete() {
   328→    if (!pendingDelete) return;
   329→    setDeleting(true);
   330→    try {
   331→      const res = await api.deleteMeal(pendingDelete.id);
   332→      toast({
   333→        title: "Meal deleted",
   334→        description: `${res.removedFoods} item${res.removedFoods === 1 ? "" : "s"} · ${formatKcal(res.removedCalories)} removed from today's totals.`,
   335→      });
   336→      setPendingDelete(null);
   337→      bumpData();
   338→    } catch (e) {
   339→      toast({
   340→        title: "Could not delete meal",
   341→        description: e instanceof Error ? e.message : "Please try again.",
   342→        variant: "destructive",
   343→      });
   344→    } finally {
   345→      setDeleting(false);
   346→    }
   347→  }
   348→
   349→  async function handleRelog() {
   350→    if (!pendingRelog) return;
   351→    setRelogging(true);
   352→    try {
   353→      const res = await api.relogMeal(pendingRelog.id);
   354→      toast({
   355→        title: "Logged again",
   356→        description: `${res.totals.calories} kcal · recalculated fresh from the food database.`,
   357→      });
   358→      setPendingRelog(null);
   359→      bumpData();
   360→    } catch (e) {
   361→      toast({
   362→        title: "Could not log this meal again",
   363→        description: e instanceof Error ? e.message : "Please try again.",
   364→        variant: "destructive",
   365→      });
   366→    } finally {
   367→      setRelogging(false);
   368→    }
   369→  }
   370→
   371→  return (
   372→    <Card className="transition-shadow duration-300 hover:shadow-md hover:shadow-primary/5">
   373→      <CardHeader>
   374→        <CardTitle className="flex items-center gap-2">
   375→          <History className="h-5 w-5 text-primary" aria-hidden />
   376→          Recent meals
   377→        </CardTitle>
   378→        <CardDescription>
   379→          Your last logged meals with server-verified totals. Expand for the full nutrient panel, or edit and delete a mistaken entry.
   380→        </CardDescription>
   381→      </CardHeader>
   382→      <CardContent>
   383→        {error && <p className="rounded-md border border-destructive/40 bg-destructive/10 px-3 py-2 text-sm text-destructive">{error}</p>}
   384→
   385→        {!data && !error && (
   386→          <div className="space-y-2">
   387→            <Skeleton className="h-16 w-full" />
   388→            <Skeleton className="h-16 w-full" />
   389→            <Skeleton className="h-16 w-full" />
   390→          </div>
   391→        )}
   392→
   393→        {data && data.count === 0 && (
   394→          <p className="rounded-xl border border-dashed p-6 text-center text-sm text-muted-foreground">
   395→            Nothing logged yet — log your first meal above and it will appear here.
   396→          </p>
   397→        )}
   398→
   399→        {data && data.count > 0 && (
   400→          <ul className="max-h-96 space-y-2 overflow-y-auto pr-1 [scrollbar-width:thin]">
   401→            {data.meals.map((m) => (
   402→              <li key={m.id}>
   403→                <Collapsible>
   404→                  <div className="group/meal relative rounded-xl border transition-colors hover:border-primary/40 hover:bg-muted/30 data-[state=open]:border-primary/40">
   405→                    <CollapsibleTrigger className="flex w-full items-center gap-3 rounded-xl p-3 pr-20 text-left transition-colors hover:bg-muted/50">
   406→                      <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-muted text-base transition-transform group-hover/meal:scale-105" aria-hidden>
   407→                        {MEAL_TYPE_ICON[m.mealType] ?? "🍽️"}
   408→                      </span>
   409→                      <span className="min-w-0 flex-1">
   410→                        <span className="flex flex-wrap items-center gap-2">
   411→                          <span className="text-sm font-semibold">{mealLabel(m.mealType)}</span>
   412→                          <span className="text-xs text-muted-foreground">{format(new Date(m.eatenAt), "d MMM, HH:mm")}</span>
   413→                          <ComplianceBadge compliance={m.compliance} conditions={healthConditions} />
   414→                          {m.source === "recommendation" && (
   415→                            <Badge variant="secondary" className="text-[9px] uppercase tracking-wide">
   416→                              from rec
   417→                            </Badge>
   418→                          )}
   419→                          {m.source === "image" && (
   420→                            <Badge variant="secondary" className="gap-0.5 text-[9px] uppercase tracking-wide">
   421→                              <Camera className="h-2.5 w-2.5" aria-hidden /> photo
   422→                            </Badge>
   423→                          )}
   424→                          {m.source === "relog" && (
   425→                            <Badge variant="secondary" className="gap-0.5 text-[9px] uppercase tracking-wide">
   426→                              <Repeat className="h-2.5 w-2.5" aria-hidden /> again
   427→                            </Badge>
   428→                          )}
   429→                        </span>
   430→                        <span className="mt-0.5 block truncate text-xs text-muted-foreground">
   431→                          {m.foods.map((f) => f.name).join(" · ")}
   432→                        </span>
   433→                      </span>
   434→                      <span className="shrink-0 text-right">
   435→                        <span className="block text-sm font-semibold tabular-nums">{formatKcal(m.totals.calories)}</span>
   436→                        <span className="block text-[10px] text-muted-foreground tabular-nums">
   437→                          P {formatGrams(m.totals.protein)} · C {formatGrams(m.totals.carbohydrates)} · F {formatGrams(m.totals.fat)}
   438→                        </span>
   439→                      </span>
   440→                      <ChevronDown className="h-4 w-4 shrink-0 text-muted-foreground transition-transform group-data-[state=open]:rotate-180" aria-hidden />
   441→                    </CollapsibleTrigger>
   442→
   443→                    {/* Edit + delete — overlay the row's right edge. Always visible on touch-sized
   444→                        screens (no hover), hover-revealed from sm up. */}
   445→                    <div className="absolute right-2 top-2 flex gap-1 opacity-100 transition-all focus-within:opacity-100 sm:opacity-0 sm:group-hover/meal:opacity-100">
   446→                      <Button
   447→                        variant="ghost"
   448→                        size="icon"
   449→                        aria-label={`Edit ${mealLabel(m.mealType)} logged at ${format(new Date(m.eatenAt), "HH:mm")}`}
   450→                        className="h-7 w-7 text-muted-foreground/50 transition-all hover:bg-primary/10 hover:text-primary"
   451→                        onClick={() => setEditing(m)}
   452→                      >
   453→                        <Pencil className="h-3.5 w-3.5" aria-hidden />
   454→                      </Button>
   455→                      <Button
   456→                        variant="ghost"
   457→                        size="icon"
   458→                        aria-label={`Delete ${mealLabel(m.mealType)} logged at ${format(new Date(m.eatenAt), "HH:mm")}`}
   459→                        className="h-7 w-7 text-muted-foreground/50 transition-all hover:bg-destructive/10 hover:text-destructive"
   460→                        onClick={() => setPendingDelete(m)}
   461→                      >
   462→                        <Trash2 className="h-3.5 w-3.5" aria-hidden />
   463→                      </Button>
   464→                    </div>
   465→
   466→                    <CollapsibleContent>
   467→                      <div className="mx-3 mb-3 mt-1 space-y-2.5 rounded-lg border bg-muted/30 p-3">
   468→                        {/* Compliance violations (if any) */}
   469→                        {m.compliance && m.compliance.violations.length > 0 && (
   470→                          <ul className="space-y-1">
   471→                            {m.compliance.violations.map((v, i) => (
