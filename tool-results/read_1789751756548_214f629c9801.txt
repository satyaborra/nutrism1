    85→interface Draft {
    86→  age: string;
    87→  sex: string;
    88→  heightCm: string;
    89→  weightKg: string;
    90→  activityLevel: string;
    91→  goal: string;
    92→  dietaryPreference: string;
    93→  allergies: string[];
    94→  healthConditions: string[];
    95→  language: string;
    96→  regional: boolean;
    97→}
    98→
    99→function draftFromServer(data: ProfileResponse): Draft {
   100→  const p = data.profile;
   101→  return {
   102→    age: p.age?.toString() ?? "",
   103→    sex: p.sex ? p.sex.charAt(0).toUpperCase() + p.sex.slice(1).toLowerCase() : "Male",
   104→    heightCm: p.heightCm?.toString() ?? "",
   105→    weightKg: p.weightKg?.toString() ?? "",
   106→    activityLevel: p.activityLevel ?? "moderate",
   107→    goal: p.goal ?? "maintain",
   108→    dietaryPreference: p.dietaryPreference ?? "vegetarian",
   109→    allergies: p.allergies ?? [],
   110→    healthConditions: p.healthConditions ?? [],
   111→    language: p.language ?? "en",
   112→    regional: (p.language ?? "en") !== "en",
   113→  };
   114→}
   115→
   116→/** Server stores sex lowercase — the UI shows display case. */
   117→function sexForServer(display: string): string {
   118→  return display.toLowerCase();
   119→}
   120→
   121→function initialsOf(name: string): string {
   122→  return name
   123→    .split(/\s+/)
   124→    .filter(Boolean)
   125→    .map((w) => w[0])
   126→    .slice(0, 2)
   127→    .join("")
   128→    .toUpperCase();
   129→}

   132→  return (

   134→      {children}
   135→    </Label>
   136→  );
   137→}

   139→function SuffixedInput({
   140→  id,
   141→  value,
   142→  onChange,
   143→  suffix,
   144→  placeholder,
   145→}: {
   146→  id: string;
   147→  value: string;
   148→  onChange: (v: string) => void;
   149→  suffix: string;
   150→  placeholder?: string;
   151→}) {
   152→  return (
   153→    <div className="relative">
   154→      <Input
   155→        id={id}
   156→        value={value}
   157→        inputMode="numeric"
   158→        placeholder={placeholder}
   159→        onChange={(e) => onChange(e.target.value.replace(/[^\d]/g, ""))}
   160→        className="h-10 rounded-xl bg-background pr-14 text-sm font-semibold"
   161→      />
   162→      <span className="pointer-events-none absolute right-3.5 top-1/2 -translate-y-1/2 text-xs font-medium text-muted-foreground">
   163→        {suffix}
   164→      </span>
   165→    </div>
   166→  );
   167→}
