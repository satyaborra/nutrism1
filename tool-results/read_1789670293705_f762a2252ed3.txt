    63→/* ---- Minimal Web Speech API typings (browser voice input, no backend) ---- */
    64→interface SpeechRecognitionEventLike {
    65→  results: ArrayLike<ArrayLike<{ transcript: string }>>;
    66→}
    67→interface SpeechRecognitionLike {
    68→  lang: string;
    69→  interimResults: boolean;
    70→  continuous: boolean;
    71→  start(): void;
    72→  stop(): void;
    73→  onresult: ((e: SpeechRecognitionEventLike) => void) | null;
    74→  onerror: ((e: { error?: string }) => void) | null;
    75→  onend: (() => void) | null;
    76→}
    77→type SpeechRecognitionCtor = new () => SpeechRecognitionLike;
    78→
    79→function getSpeechRecognition(): SpeechRecognitionCtor | null {
    80→  if (typeof window === "undefined") return null;
    81→  const w = window as unknown as { SpeechRecognition?: SpeechRecognitionCtor; webkitSpeechRecognition?: SpeechRecognitionCtor };
    82→  return w.SpeechRecognition ?? w.webkitSpeechRecognition ?? null;
    83→}

    93→function dayKeyOf(d: Date): string {
    94→  const y = d.getFullYear();
    95→  const m = String(d.getMonth() + 1).padStart(2, "0");
    96→  const day = String(d.getDate()).padStart(2, "0");
    97→  return `${y}-${m}-${day}`;
    98→}
    99→
   100→function todayKey(): string {
   101→  return dayKeyOf(new Date());
   102→}
   103→
   104→function yesterdayKey(): string {
   105→  const d = new Date();
   106→  d.setDate(d.getDate() - 1);
   107→  return dayKeyOf(d);
   108→}
   109→
   110→function minBackfillKey(): string {
   111→  const d = new Date();
   112→  d.setDate(d.getDate() - 30);
   113→  return dayKeyOf(d);
   114→}
   115→
   116→function prettyDate(key: string): string {
   117→  return new Date(`${key}T12:00:00`).toLocaleDateString("en-GB", { weekday: "short", day: "numeric", month: "short" });
   118→}

   183→  function resetAll() {
   184→    setStep("input");
   185→    setText("");
   186→    setImageDataUrl(null);
   187→    setImageHint("");
   188→    setMeta(null);
   189→    setLines([]);
   190→    setConfirmResult(null);
   191→    setConfirmError(null);
   192→    setAnalyzeError(null);
   193→    setLogDate(todayKey());
   194→    clearBackfill();

   199→  useEffect(() => {
   200→    if (!backfillRequest) return;
   201→    setStep("input");
   202→    setLogDate(backfillRequest.date);

   210→    setStep("input");

   212→    if (loggerRequest.openFile) {
   213→      // wait a tick so the photo tab content (and its file input) exists
   214→      requestAnimationFrame(() => fileInputRef.current?.click());
   215→    }
   216→    if (loggerRequest.voice) {
   217→      requestAnimationFrame(() => startVoiceInput());
   218→    }
   219→    clearLoggerRequest();

   222→  /** Voice input via the browser Web Speech API — transcript lands in the
   223→   *  describe textarea; the AI pipeline handles the rest. */
   224→  function startVoiceInput() {
   225→    const Ctor = getSpeechRecognition();
   226→    if (!Ctor) {
   227→      toast({
   228→        title: "Voice input not available",
   229→        description: "Your browser doesn't support speech recognition — try Chrome, or type the meal instead.",
   230→        variant: "destructive",
   231→      });
   232→      return;
   233→    }
   234→    if (voiceRef.current) {
   235→      voiceRef.current.stop();
   236→      voiceRef.current = null;
   237→      setVoiceListening(false);
   238→      return;
   239→    }
   240→    try {
   241→      const rec = new Ctor();
   242→      rec.lang = "en-IN";
   243→      rec.interimResults = false;
   244→      rec.continuous = false;
   245→      rec.onresult = (e) => {
   246→        const transcript = Array.from({ length: e.results.length }, (_, i) => e.results[i][0]?.transcript ?? "")
   247→          .join(" ")
   248→          .trim();
   249→        if (transcript) {
   250→          setText((prev) => (prev ? `${prev} ${transcript}` : transcript));
   251→          setTab("text");
   252→          toast({ title: "Heard you", description: `“${transcript}” — check the text, then Analyze.` });
   253→        }
   254→      };
   255→      rec.onerror = (e) => {
   256→        toast({ title: "Could not listen", description: e.error === "not-allowed" ? "Microphone permission was denied." : "Try again in a quieter moment.", variant: "destructive" });
   257→        setVoiceListening(false);
   258→        voiceRef.current = null;
   259→      };
   260→      rec.onend = () => {
   261→        setVoiceListening(false);
   262→        voiceRef.current = null;
   263→      };
   264→      voiceRef.current = rec;
   265→      setVoiceListening(true);
   266→      rec.start();
   267→      toast({ title: "Listening…", description: "Speak your meal — e.g. “two idli and one cup sambar”." });
   268→    } catch {
   269→      setVoiceListening(false);
   270→      voiceRef.current = null;
   271→      toast({ title: "Could not start the microphone", description: "Please allow microphone access and try again.", variant: "destructive" });
   272→    }
   273→  }

   275→  async function handleAnalyze() {
   276→    setAnalyzing(true);
   277→    setAnalyzeError(null);
   278→    try {
   279→      const payload: { text?: string; imageDataUrl?: string; hint?: string } = {};
   280→      if (tab === "text" && text.trim()) payload.text = text.trim();
   281→      if (tab === "photo" && imageDataUrl) {
   282→        payload.imageDataUrl = imageDataUrl;
   283→        if (imageHint.trim()) payload.hint = imageHint.trim();
   284→      }
   285→      const res = await api.analyzeFood(payload);
   286→      setMeta({
   287→        draftId: res.draftId,
   288→        inputType: tab === "photo" ? "image" : "text",
   289→        detectedLanguage: res.detectedLanguage,
   290→        aiOk: res.aiOk,
   291→        aiNote: res.aiNote,
   292→        aiLatencyMs: res.aiLatencyMs,
   293→      });
   294→      setLines(
   295→        res.foods.map((f) => ({
   296→          lineId: f.lineId,
   297→          originalName: f.originalName,
   298→          displayName: f.displayName,
   299→          foodId: f.foodId,
   300→          matchStatus: f.matchStatus,
   301→          candidates: f.candidates,
   302→          quantity: f.quantity,
   303→          unit: f.unit,
   304→          preparation: f.preparation ?? null,
   305→          confidence: f.confidence,
   306→          quantitySource: f.quantitySource,
   307→        }))
   308→      );
   309→      setMealType(res.mealTypeGuess || "snack");
   310→      setConfirmResult(null);
   311→      setStep("review");
   312→      if (!res.aiOk) {
   313→        toast({
   314→          title: "Parsed offline",
   315→          description: "The AI service was unavailable, so a fallback parser was used. Please double-check the items below.",
   316→          variant: "destructive",
   317→        });
   318→      }
   319→    } catch (e) {
   320→      setAnalyzeError(e instanceof ApiError ? e.message : "Analysis failed. Please try again.");
   321→    } finally {
   322→      setAnalyzing(false);
   323→    }
   324→  }

   391→  async function handleLog() {
   392→    setLogging(true);
   393→    try {
   394→      // Backfilled meals get a synthetic slot time on the chosen day; today
   395→      // logs live at "now" exactly as before.

   397→      const res = await api.logMeal({
   398→        requestId: crypto.randomUUID(),
   399→        draftId: meta?.draftId,
   400→        mealType,
   401→        foods: foodsPayload(),
   402→        // Image-based drafts are labeled "image" so the history shows how it was logged
   403→        source: meta?.inputType === "image" ? "image" : meta?.inputType === "image_text" ? "image" : "text",
   404→        ...(isBackfill ? { eatenAt: `${logDate}T${SLOT_DEFAULT_TIME[mealType] ?? "12:30"}:00` } : {}),
   405→      });
   406→      toast({
   407→        title: res.duplicate ? "Meal already logged" : isBackfill ? `Logged for ${prettyDate(logDate)}` : "Meal logged",
   408→        description: `${mealLabel(mealType)} · ${formatKcal(res.totals.calories)}${isBackfill ? " · added to that day's history" : ""}${res.compliance.state !== "COMPLIANT" ? " · check the alerts below" : ""}`,
   409→      });
   410→      onLogged();
   411→      setLoggedResult({
   412→        mealType,
   413→        duplicate: res.duplicate,
   414→        totals: res.totals,
   415→        compliance: res.compliance,
   416→        backfillDate: isBackfill ? logDate : null,
   417→      });
   418→      setStep("logged");
   419→    } catch (e) {
   420→      toast({
   421→        title: "Could not log meal",
   422→        description: e instanceof ApiError ? e.message : "Unexpected error. Please try again.",
   423→        variant: "destructive",
   424→      });
   425→    } finally {
   426→      setLogging(false);
   427→    }
   428→  }

   777→/** True when lines were edited after the confirm response was produced. */
   778→function isDirtyAgainst(lines: EditableLine[], res: ConfirmFoodResponse): boolean {
   779→  if (res.foods.length !== lines.length) return true;
   780→  return lines.some((l, i) => {
   781→    const r = res.foods[i];
   782→    return (
   783→      r.lineId !== l.lineId ||
   784→      r.quantity !== Number(l.quantity) ||
   785→      r.unit !== l.unit ||
   786→      (r.foodId ?? null) !== l.foodId
   787→    );
   788→  });
   789→}

   915→function FoodSearchInline({ onPick, onCancel }: { onPick: (f: FoodSearchItem) => void; onCancel: () => void }) {
   916→  const [q, setQ] = useState("");
   917→  const [results, setResults] = useState<FoodSearchItem[]>([]);
   918→  const [loading, setLoading] = useState(false);
   919→  const [touched, setTouched] = useState(false);
   920→
   921→  // Debounced search against /api/foods/search
   922→  async function run(query: string) {
   923→    if (query.trim().length < 2) {
   924→      setResults([]);
   925→      return;
   926→    }
   927→    setLoading(true);
   928→    try {
   929→      const res = await fetch(`/api/foods/search?q=${encodeURIComponent(query.trim())}`, { cache: "no-store" });
   930→      if (res.ok) {
   931→        const body = (await res.json()) as { foods: FoodSearchItem[] };
   932→        setResults(body.foods.slice(0, 8));
   933→      }
   934→    } finally {
   935→      setLoading(false);
   936→    }
   937→  }
   938→
   939→  return (
   940→    <div className="rounded-lg border bg-background p-2">
   941→      <div className="flex items-center gap-2">
   942→        <Search className="h-4 w-4 shrink-0 text-muted-foreground" aria-hidden />
   943→        <Input
   944→          autoFocus
   945→          aria-label="Search food database"
   946→          placeholder="Search foods… (English or native names)"
   947→          className="h-8 border-none shadow-none focus-visible:ring-0"
   948→          value={q}
   949→          onChange={(e) => {
   950→            setQ(e.target.value);
   951→            setTouched(true);
   952→            void run(e.target.value);
   953→          }}
   954→          onKeyDown={(e) => e.key === "Escape" && onCancel()}
   955→        />
   956→        {loading && <Loader2 className="h-4 w-4 shrink-0 animate-spin text-muted-foreground" aria-hidden />}
   957→        <Button variant="ghost" size="icon" className="h-7 w-7 shrink-0" aria-label="Close search" onClick={onCancel}>
   958→          <X className="h-3.5 w-3.5" />
   959→        </Button>
   960→      </div>
   961→      {q.trim().length >= 2 && results.length > 0 && (
   962→        <ul className="mt-1 max-h-44 space-y-0.5 overflow-y-auto">
   963→          {results.map((f) => (
   964→            <li key={f.id}>
   965→              <button
   966→                type="button"
   967→                onClick={() => onPick(f)}
   968→                className="flex w-full items-center gap-2 rounded-md px-2 py-1.5 text-left text-sm transition-colors hover:bg-muted"
   969→              >
   970→                <span className="min-w-0 flex-1 truncate font-medium">{f.name}</span>
   971→                <span className="shrink-0 text-[10px] uppercase tracking-wide text-muted-foreground">{f.category}</span>
   972→                {f.isVeg && (
   973→                  <span className="flex h-3.5 w-3.5 shrink-0 items-center justify-center rounded-[3px] border border-emerald-600" title="Vegetarian">
   974→                    <span className="h-1.5 w-1.5 rounded-full bg-emerald-600" />
   975→                  </span>
   976→                )}
   977→              </button>
   978→            </li>
   979→          ))}
   980→        </ul>
   981→      )}
   982→      {touched && q.trim().length >= 2 && !loading && results.length === 0 && (
   983→        <p className="px-2 py-1.5 text-xs text-muted-foreground">No matches — you can keep it unmatched (it will log with 0 nutrition).</p>
   984→      )}
   985→    </div>
   986→  );
