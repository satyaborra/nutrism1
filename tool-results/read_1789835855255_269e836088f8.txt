    80→type FieldKey = "age" | "heightCm" | "weightKg";

    82→const FIELD_LIMITS: Record<FieldKey, { min: number; max: number; message: string }> = {
    83→  age: { min: 5, max: 120, message: "Age must be between 5 and 120." },
    84→  heightCm: { min: 80, max: 250, message: "Height must be between 80 and 250 cm." },
    85→  weightKg: { min: 15, max: 400, message: "Weight must be between 15 and 400 kg." },

    88→const FIELD_ORDER: FieldKey[] = ["age", "heightCm", "weightKg"];

    90→/** Keep digits and a single decimal dot; drop everything else. */
    91→function sanitizeNumeric(v: string): string {
    92→  let out = "";
    93→  let dot = false;
    94→  for (const ch of v) {
    95→    if (ch >= "0" && ch <= "9") out += ch;
    96→    else if (ch === "." && !dot) {
    97→      out += ".";
    98→      dot = true;
    99→    }
   100→  }
   101→  return out;
   102→}
   103→
   104→function validateDraft(draft: Record<FieldKey, string>): Partial<Record<FieldKey, string>> {
   105→  const errors: Partial<Record<FieldKey, string>> = {};
   106→  for (const key of FIELD_ORDER) {
   107→    const raw = draft[key].trim();
   108→    if (raw === "") continue; // empty = clear the value (null is allowed server-side)
   109→    const n = Number(raw);
   110→    if (!Number.isFinite(n) || n < FIELD_LIMITS[key].min || n > FIELD_LIMITS[key].max) {
   111→      errors[key] = FIELD_LIMITS[key].message;
   112→    }
   113→  }
   114→  return errors;
   115→}

   142→  inputId?: string;
   143→  editValue?: string;
   144→  error?: string;
   145→  onChange?: (v: string) => void;
   146→  fieldRef?: (el: HTMLDivElement | null) => void;
   147→  disabled?: boolean;
   148→}) {
   149→  return (
   150→    <div ref={fieldRef} className="scroll-mt-24 border-b border-dashed border-border/60 py-2.5 last:border-b-0">
   151→      <div className="flex items-center justify-between gap-3">

   153→          <span
   154→            aria-hidden
   155→            className={cn(

   157→              iconTone ?? "bg-primary/10 text-primary",
   158→            )}
   159→          >
   160→            {icon}
   161→          </span>
   162→          <span className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">{label}</span>
   163→        </div>
   164→        {inputId ? (
   165→          <div className="flex flex-col items-end gap-1">
   166→            <div className="flex items-center gap-1.5">
   167→              <Input
   168→                id={inputId}
   169→                inputMode="decimal"
   170→                autoComplete="off"
   171→                value={editValue ?? ""}
   172→                onChange={(e) => onChange?.(e.target.value)}
   173→                aria-invalid={error ? true : undefined}
   174→                aria-describedby={error ? `${inputId}-error` : undefined}
   175→                disabled={disabled}
   176→                className="h-10 w-24 text-right text-sm font-semibold tabular-nums"
   177→              />
   178→              <span className="w-10 text-xs text-muted-foreground">{unit}</span>
   179→            </div>
   180→            {error && (
   181→              <p id={`${inputId}-error`} role="alert" className="max-w-56 text-right text-xs font-medium text-destructive">
   182→                {error}
   183→              </p>
   184→            )}
   185→          </div>
   186→        ) : (
   187→          <div className="text-right">
   188→            {value != null ? (
   189→              <span className="text-lg font-bold tabular-nums text-emerald-950 dark:text-emerald-50">
   190→                {value}
   191→                {unit ? <span className="ml-1 text-xs font-medium text-muted-foreground">{unit}</span> : null}
   192→              </span>
   193→            ) : (
   194→              <span className="text-lg font-bold text-muted-foreground/50">
   195→                — <span className="text-xs font-medium">not set</span>
   196→              </span>
   197→            )}
   198→          </div>
   199→        )}
   200→      </div>
   201→    </div>
   202→  );
   203→}
