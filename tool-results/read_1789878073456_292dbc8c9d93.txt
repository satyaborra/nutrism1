    62→/** Shift a YYYY-MM-DD key by N days (local time, noon-anchored). */
    63→function shiftDayKey(key: string, days: number): string {
    64→  return format(addDays(parseISO(`${key}T12:00:00`), days), "yyyy-MM-dd");
    65→}
    66→
    67→/** "08:30 AM" from an ISO datetime — never crashes on bad input. */
    68→function timeLabel(iso: string): string {
    69→  const d = parseISO(iso);
    70→  return Number.isNaN(d.getTime()) ? "—" : format(d, "hh:mm a");
    71→}
    72→
    73→/** "Fri 19 Sep, 08:30 AM" from an ISO datetime. */
    74→function dayTimeLabel(iso: string): string {
    75→  const d = parseISO(iso);
    76→  return Number.isNaN(d.getTime()) ? "—" : format(d, "EEE d MMM, hh:mm a");
    77→}

    91→/**
    92→ * Honest provenance badge. The API's meal.source is "text" | "manual" |
    93→ * "image" | "recommendation" | "relog" — there is no "verified"/"photo"
    94→ * value, so: "Verified ✓" only when every food line matched the verified DB
    95→ * food library (foodId non-null); "From Scan" for photo-logged meals that
    96→ * are not fully matched. Everything else shows no badge.
    97→ */
    98→function SourceBadge({ meal }: { meal: MealDetail }) {
    99→  const allMatched = meal.foods.length > 0 && meal.foods.every((f) => f.foodId !== null);
   100→  if (allMatched) {
   101→    return (
   102→      <Badge variant="outline" className="border-emerald-500/40 bg-emerald-500/10 px-1.5 py-0 text-[10px] font-medium text-emerald-700 dark:text-emerald-400">
   103→        Verified ✓
   104→      </Badge>
   105→    );
   106→  }
   107→  if (meal.source === "image") {
   108→    return (
   109→      <Badge variant="secondary" className="px-1.5 py-0 text-[10px] font-medium text-muted-foreground">
   110→        From Scan
   111→      </Badge>
   112→    );
   113→  }
   114→  return null;

   127→function MealsHeroStats({ hero }: { hero: HeroData }) {
   128→  const { today, yesterday, week } = hero;
   129→  const kcal = today.consumed.calories;
   130→  const target = today.targets.calories;
   131→  const mealsLogged = today.meals.length;
   132→
   133→  // vs-yesterday delta — only when a comparable baseline exists.
   134→  const yKcal = yesterday?.consumed.calories ?? null;
   135→  let deltaSub: ReactNode = "First logged day";
   136→  let deltaTone: "muted" | "up" | "down" = "muted";
   137→  if (yKcal !== null) {
   138→    if (yKcal <= 0) {
   139→      deltaSub = "No intake logged yesterday";
   140→    } else {
   141→      const delta = Math.round(((kcal - yKcal) / yKcal) * 100);
   142→      if (delta > 0) {
   143→        deltaSub = `↑ ${delta}% vs yesterday`;
   144→        deltaTone = "up";
   145→      } else if (delta < 0) {
   146→        deltaSub = `↓ ${Math.abs(delta)}% vs yesterday`;
   147→        deltaTone = "down";
   148→      } else {
   149→        deltaSub = "Same as yesterday";
   150→      }
   151→    }
   152→  }
   153→
   154→  return (
   155→    <>
   156→      <StatCard
   157→        icon={<Flame className="h-4 w-4" aria-hidden />}
   158→        tone="rose"
   159→        label="Total Intake"
   160→        value={formatKcal(kcal)}
   161→        sub={deltaSub}
   162→        subTone={deltaTone}
   163→        extra={<Sparkline data={week.days.map((d) => d.calories)} tone="rose" />}
   164→      />
   165→      <StatCard
   166→        icon={<Target className="h-4 w-4" aria-hidden />}
   167→        tone="emerald"
   168→        label="Daily Target"
   169→        value={formatKcal(target)}
   170→        progress={target > 0 ? kcal / target : 0}
   171→        sub={`${pct(kcal, target)}% achieved`}
   172→      />
   173→      <StatCard
   174→        icon={<Leaf className="h-4 w-4" aria-hidden />}
   175→        tone="teal"
   176→        label="Meals Logged"
   177→        value={`${mealsLogged} / 5`}
   178→        extra={
   179→          <span className="flex items-center gap-1" role="img" aria-label={`${mealsLogged} of 5 meal slots logged today`}>
   180→            {Array.from({ length: 5 }).map((_, i) => (
   181→              <span key={i} aria-hidden className={cn("h-2.5 w-2.5 rounded-full", i < mealsLogged ? "bg-primary" : "bg-primary/20")} />
   182→            ))}
   183→          </span>
   184→        }
   185→      />
   186→      <StatCard
   187→        icon={<Sparkles className="h-4 w-4" aria-hidden />}
   188→        tone="amber"
   189→        label="Streak"
   190→        value={`${week.streak} ${week.streak === 1 ? "day" : "days"}`}
   191→        sub="Keep it going!"
   192→      />
   193→    </>
   194→  );
   195→}

   399→export function MealsView() {
   400→  const dataVersion = useNutriStore((s) => s.dataVersion);
   401→  const setView = useNutriStore((s) => s.setView);
   402→  const bumpData = useNutriStore((s) => s.bumpData);
   403→  const { toast } = useToast();
   404→
   405→  const TODAY = todayKey();
   406→  const MIN_DATE = shiftDayKey(TODAY, -365);
   407→
   408→  const [tab, setTab] = useState<PeriodTab>("today");
   409→  const [selectedDate, setSelectedDate] = useState<string>(TODAY);
   410→  const [hero, setHero] = useState<HeroData | null>(null);
   411→  const [heroFailed, setHeroFailed] = useState(false);
   412→  const [day, setDay] = useState<{ date: string; summary: DailySummaryResponse | null } | null>(null);
   413→  const [recent, setRecent] = useState<MealDetail[] | null>(null);
   414→  const [recentFailed, setRecentFailed] = useState(false);
   415→  const [expandedId, setExpandedId] = useState<string | null>(null);
   416→  const [waterBusy, setWaterBusy] = useState(false);
   417→
   418→  // Hero stats + the selected day's timeline, in parallel. Re-fetched on
   419→  // every dataVersion bump (meal edits/deletes) and day change. Failure is
   420→  // silent (console.warn) — the hero renders without stats, sections stay up.
   421→  useEffect(() => {
   422→    let alive = true;
   423→    const tKey = todayKey();
   424→    Promise.all([
   425→      api.dailySummary(tKey),
   426→      api.dailySummary(shiftDayKey(tKey, -1)),
   427→      api.weeklySummary(),
   428→      selectedDate === tKey ? Promise.resolve(null) : api.dailySummary(selectedDate),
   429→    ])
   430→      .then(([today, yesterday, week, selected]) => {
   431→        if (!alive) return;
   432→        setHero({ today, yesterday, week });
   433→        setHeroFailed(false);
   434→        setDay({ date: selectedDate, summary: selected ?? today });
   435→      })
   436→      .catch((err) => {
   437→        console.warn("[meals] summaries unavailable:", err);
   438→        if (!alive) return;
   439→        setHeroFailed(true);
   440→        setDay((prev) => (prev && prev.date === selectedDate ? prev : { date: selectedDate, summary: null }));
   441→      });
   442→    return () => {
   443→      alive = false;
   444→    };
   445→  }, [dataVersion, selectedDate]);
   446→
   447→  // Recent meals for the bottom strip (independent — failures don't take
   448→  // the rest of the view down).
   449→  useEffect(() => {
   450→    let alive = true;
   451→    api
   452→      .recentMeals()
   453→      .then((res) => {
   454→        if (!alive) return;
   455→        setRecent(res.meals.slice(0, 4));
   456→        setRecentFailed(false);
   457→      })
   458→      .catch((err) => {
   459→        console.warn("[meals] recent meals unavailable:", err);
   460→        if (!alive) return;
   461→        setRecentFailed(true);
   462→      });
   463→    return () => {
   464→      alive = false;
   465→    };
   466→  }, [dataVersion]);
   467→
   468→  function shiftDate(days: number) {
   469→    setSelectedDate((prev) => shiftDayKey(prev, days));
   470→  }
   471→
   472→  async function quickAddWater() {
   473→    if (waterBusy) return;
   474→    setWaterBusy(true);
   475→    try {
   476→      await api.hydrationUpdate({ delta: 1 });
   477→      bumpData();
   478→      toast({ title: "+1 glass logged", description: "Your water intake was updated." });
   479→    } catch {
   480→      toast({ title: "Could not log water", description: "Please try again in a moment.", variant: "destructive" });
   481→    } finally {
   482→      setWaterBusy(false);
   483→    }
   484→  }
   485→
   486→  // Timeline data for the selected day: undefined = still loading, null = failed.
   487→  const activeDay = day && day.date === selectedDate ? day.summary : undefined;
   488→  const meals = activeDay?.meals ?? [];
   489→
   490→  const quickAdds: { emoji: string; label: string; aria: string; onClick: () => void; disabled?: boolean }[] = [
   491→    { emoji: "💧", label: "Water", aria: "Quick add a glass of water", onClick: quickAddWater, disabled: waterBusy },
   492→    { emoji: "🍎", label: "Fruits", aria: "Log fruits in the food logger", onClick: () => setView("log") },
   493→    { emoji: "🥜", label: "Snacks", aria: "Log snacks in the food logger", onClick: () => setView("log") },
   494→    { emoji: "➕", label: "Custom", aria: "Log a custom meal in the food logger", onClick: () => setView("log") },
   495→  ];
   496→
   497→  return (
   498→    <div className="space-y-5">
   499→      {/* ---------------- HERO ---------------- */}
